aa<-ggplot() +
  geom_bar(
    data = data_melted,
    aes(x = Traits, y = Value, fill = Parameter),
    stat = "identity",
    position = position_dodge(width = 0.8),
    show.legend = TRUE
  ) +
  geom_point(
    data = data,
    aes(x = Traits, y = h2, color = "Heritability (h²)"),
    size = 3,
    shape = 18  # Diamond shape for points
  ) +
  geom_text(
    data = data_melted,
    aes(x = Traits, y = Value, label = ifelse(Value %% 10 == 0, Value, "")),  # Show labels only for multiples of 10
    position = position_dodge(width = 0.8),
    vjust = -0.5,  # Adjust text placement vertically above the bars
    size = 4, 
    family = "Times New Roman",
    color = "black"
  ) +
  scale_fill_manual(
    values = c("#FF6F61", "#4A90E2", "#F5A623"),  # Brighter colors for GCV, PCV, GM
    name = "Parameters", 
    labels = c("GCV (%)", "PCV (%)", "GM (%)")  # Custom labels for the legend
  ) +
  scale_color_manual(
    values = c("Heritability (h²)" = "black"),  # Custom color for Heritability points
    name = NULL
  ) +
  scale_y_continuous(breaks = seq(0, 100, by = 10)) +  # Custom y-axis breaks
  labs(
    x = "Quantitative traits",
    y = "Percentage (%)"
  ) +
  theme_minimal(base_size = 14) +  # Increase base font size for readability
  theme(
    axis.text.x = element_text(
      angle = 45, 
      hjust = 1, 
      size = 12, 
      color = "black",  # Black color for x-axis labels
      family = "Times New Roman"
    ),
    axis.text.y = element_text(
      size = 12, 
      color = "black",  # Black color for y-axis tick labels
      family = "Times New Roman"
    ),
    axis.title.x = element_text(color = "black", size = 14, family = "Times New Roman"),  # Set font to Times New Roman
    axis.title.y = element_text(color = "black", size = 14, family = "Times New Roman"),  # Set font to Times New Roman
    plot.title = element_text(hjust = 0.5, size = 18, face = "bold", family = "Times New Roman"),
    legend.position = "right",  # Legend moved to the right side
    legend.title = element_blank(),
    legend.text = element_text(size = 10, family = "Times New Roman"),  # Times New Roman for legend text
    legend.key.size = unit(0.3, "cm"),  # Smaller legend keys
    legend.box.background = element_rect(color = "black", size = 0.5),  # Box around legend
    panel.grid.major = element_line(color = "gray90", size = 0.5),
    panel.grid.minor = element_blank()
  )
aa

ggsave(filename = "Heritabuilidgfsc345dty.jpg", plot= aa, width = 33, height = 25, dpi = 800, units = "cm")
