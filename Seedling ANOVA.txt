####### Testing for Anova assumption ######
datamodel1 <- lm(SD$RL ~ SD$GEN+SD$TRT+SD$REP)
ggqqplot(residuals(datamodel1))
shapiro.test(residuals(datamodel1))

###Homogenity of variance#####
SD %>% levene_test(SD$RL ~ GEN)

####Test for anova assumption###
gvlma_result1 <- gvlma(datamodel1)
summary(gvlma_result1)
###IF Normalization Required#####
SD$log_RL <- log(SD$RL)  # Using natural log (ln)
shapiro.test(SD$log_RL)
###ANOVA#####
datamodel <- lm(SD$RL~REP+TRT+GEN:TRT+GEN+REP,data=SD)
anova_model <- aov(datamodel)
summary(anova_model)
