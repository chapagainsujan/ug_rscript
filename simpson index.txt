# Load required libraries
library(ggplot2)
library(dplyr)

# Create the dataset using your table's data
data <- data.frame(
  Trait = c(
    "Plant Growth Habit", "Blast Resistance", "Pigmentation at Leaf Juncture", 
    "Stem Clum Branching", "Finger Branching", "Position of Finger", 
    "Finger Multiple Whorl Branching", "Spikelet Density", "Lodging Susceptibility", 
    "Inflorescence Colour", "Anther Colour", "Node Pigmentation", 
    "Leaf Blade Intensity of Colour", "Leaf Sheath Pubescence", "Leaf Blade Pubescence", 
    "Ear Shape", "Glume Colour", "Spikelet Discontinuity", 
    "Ear Size", "Grain Covering", "Grain per Spikelet", 
    "Synchrony of Ear Maturity", "Spikelet Shattering", 
    "Pericarp Persistence after Threshing", "Grain Intensity of Colour", 
    "Grain Shape", "Grain Surface", "Grain Uniformity", 
    "Grain Colour", "Plant Total Bushing"
  ),
  Simpson_Index = c(
    0.000, 0.287, 0.591, 0.485, 0.485, 0.474, 0.474, 0.368, 0.439, 
    0.544, 0.719, 0.544, 0.304, 0.000, 0.415, 0.193, 0.287, 0.474, 
    0.345, 0.368, 0.632, 0.591, 0.474, 0.474, 0.322, 0.497, 0.649, 
    0.649, 0.251, 0.567
  )
)

# Sort the data by Simpson Index for better visualization
data <- data %>%
  arrange(desc(Simpson_Index))

# Define custom color palette
custom_colors <- c("#F26C4F", "#2B7DBB", "#5FAACB", "#8DC6D9", "#A2D9D9", 
                   "#F2B134", "#F28705")

# Create the bar plot
simpson_plot <- ggplot(data, aes(x = reorder(Trait, Simpson_Index), y = Simpson_Index, fill = Simpson_Index)) +
  geom_bar(stat = "identity", width = 0.7, color = "black") +  # Add black border to bars
  coord_flip() +  # Flip coordinates for horizontal bars
  scale_fill_gradientn(colors = custom_colors) +  # Custom gradient colors
  labs(
    x = "Qualitative Trait",
    y = "Simpson Index",
    
  ) +
  theme_gray(base_size = 14) +  # Set a grey professional theme
  theme(
    text = element_text(family = "Times New Roman"),  # Use Times New Roman font
    axis.text.y = element_text(size = 12),
    axis.text.x = element_text(size = 12),
    plot.title = element_text(hjust = 0.5, size = 18, face = "bold"),  # Center-align the title
    panel.border = element_rect(color = "black", fill = NA, linewidth = 1),  # Add border
    panel.background = element_rect(fill = "grey95"),  # Light grey background
    panel.grid.major = element_line(color = "grey80", linetype = "dashed"),  # Dashed grid lines
    legend.position = "right"  # Place legend on the right
  ) +
  geom_text(aes(label = round(Simpson_Index, 3)), hjust = 1.1, size = 4, color = "black")  # Add data labels

# Save the plot
ggsave(filename = "Simpson_Ind223ex_Plot.png", plot = simpson_plot, width = 30, height = 23, dpi = 300, units = "cm")
