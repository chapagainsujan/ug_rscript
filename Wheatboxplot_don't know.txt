# ── 1. Load Required Libraries ───────────────────────────────
library(ggplot2)
library(patchwork)

# ── 2. Empty Plot Function ──────────────────────────────────
empty_plot <- function() {
  ggplot() + theme_void()
}

# ── 3. Set Improved Theme ───────────────────────────────────
theme_set(
  theme_bw(base_family = "Times New Roman", base_size = 12) +
    theme(
      panel.grid.major = element_blank(),
      panel.grid.minor = element_blank(),
      panel.border = element_rect(colour = "black", size = 1),
      axis.line = element_line(color = "black", size = 0.5),
      axis.ticks = element_line(color = "black"),
      axis.text = element_text(color = "black", size = 10),
      axis.title = element_text(size = 12),
      plot.margin = margin(5, 5, 5, 5)
    )
)

# ── 4. Function to Create Boxplots ───────────────────────────
create_boxplot <- function(data, var_name, x_label, y_units) {
  mean_val <- mean(data[[var_name]], na.rm = TRUE)
  aggregated_data <- aggregate(as.formula(paste(var_name, "~ Landraces")), data = data, FUN = mean)
  
  ggplot(data, aes(x = "", y = .data[[var_name]])) +
    geom_boxplot(color = "#00a6b3", fill = "white", outlier.shape = NA, size = 1.2, linewidth = 1.1) +
    geom_point(data = aggregated_data, aes(x = "", y = .data[[var_name]]), shape = 18, color = "#00a6b3", size = 2.7) +
    annotate("point", x = 1, y = mean_val, shape = 16, color = "black", size = 2.2) +
    xlab(x_label) + ylab(y_units) +
    theme(
      axis.text.x = element_blank(),
      axis.ticks.x = element_blank()
    )
}

# ── 5. Define Traits ─────────────────────────────────────────
vars <- list(
  list(var = "PH", xlab = "Plant Height (PH)", unit = "cm"),
  list(var = "ET", xlab = "Effective Tillers (ET)", unit = "number (n)"),
  list(var = "FLA", xlab = "Flag Leaf Area (FLA)", unit = "cm²"),
  list(var = "DTF", xlab = "Days to 50% Flowering (DTF)", unit = "days"),
  list(var = "DTM", xlab = "Days to 80% Maturity (DTM)", unit = "days"),
  list(var = "EHL", xlab = "Ear Head Length (EHL)", unit = "cm"),
  list(var = "FW", xlab = "Finger Width (FW)", unit = "cm"),
  list(var = "NF", xlab = "Number of Fingers (NF)", unit = "number (n)"),
  list(var = "EHW", xlab = "Ear Head Weight (EHW)", unit = "g"),
  list(var = "TSW", xlab = "Thousand Seed Weight (TSW)", unit = "g"),
  list(var = "YPH", xlab = "Yield per Head (YPH)", unit = "g"),
  list(var = "GY", xlab = "Grain Yield (GY)", unit = "ton/ha"),
  list(var = "BY", xlab = "Biological Yield (BY)", unit = "ton/ha"),
  list(var = "HI", xlab = "Harvest Index (HI)", unit = "%")
)

# ── 6. Generate Boxplots ─────────────────────────────────────
plots <- lapply(vars, function(v) {
  create_boxplot(Millet, v$var, v$xlab, v$unit)
})

# ── 7. Add empty plots if less than 16 ───────────────────────
while(length(plots) < 16) {
  plots <- append(plots, list(empty_plot()))
}

# ── 8. Arrange in 4x4 grid with tags outside ─────────────────
final_plot <- wrap_plots(plots, ncol = 4, nrow = 4) +
  plot_annotation(
    tag_levels = "a",          # tags a, b, c, ...
    tag_prefix = "",           # no extra prefix, just letters
    tag_fontface = "bold",
    tag_size = 18,             # <-- adjust this number for tag size
    theme = theme(text = element_text(family = "Times New Roman"))
  )

# ── 9. Save Figure ───────────────────────────────────────────
ggsave(
  filename = "Box_plot_millet_4x4_tagged_outside.tiff",
  plot = final_plot,
  width = 30,
  height = 30,
  dpi = 600,
  units = "cm",
  device = "tiff",
  compression = "lzw"
)
