library(scales)
library(GGally)
library(ggplot2)
library(dplyr)
library(grid)

SD <- SAAAAA  # Your dataset

# ── Custom Correlation Function ──
custom_cor_sig <- function(data, mapping, method = "pearson", digits = 2, ...) {
  x <- eval_data_col(data, mapping$x)
  y <- eval_data_col(data, mapping$y)
  trt <- data$TRT
  trt_levels <- levels(factor(trt))
  colors <- hue_pal()(length(trt_levels))
  names(colors) <- trt_levels
  
  results <- sapply(trt_levels, function(g) {
    idx <- which(trt == g)
    test <- suppressWarnings(cor.test(x[idx], y[idx], method = method))
    r <- round(test$estimate, digits)
    p <- test$p.value
    stars <- if (is.na(p)) "" else if (p < 0.001) "***" else if (p < 0.01) "**" else if (p < 0.05) "*" else ""
    paste0(r, stars)
  })
  
  # Adjust spacing dynamically
  n <- length(results)
  gap <- 0.5  # Vertical gap between correlation values
  start_y <- 1.6 + (gap * (n-2)/2)  # Adjust starting Y based on number of groups
  
  label_df <- data.frame(
    x = 1,
    y = seq(from = start_y, by = -gap, length.out = n),
    label = results,
    color = colors[trt_levels]
  )
  
  ggplot() +
    geom_text(data = label_df, aes(x = x, y = y, label = label, color = color),
              size = 6.5, fontface = "bold") +
    scale_color_identity() +
    coord_cartesian(xlim = c(0.5, 1.5), ylim = c(0.5, 2)) +
    theme_void()
}

# ── Prepare Data ──
trait_data <- SD %>%
  select(where(is.numeric), -REP)
trait_data$TRT <- SD$TRT
traits <- colnames(trait_data)[1:(ncol(trait_data) - 1)]

# ── Create ggpairs Plot ──
cor_plot <- ggpairs(
  data = trait_data,
  columns = 1:(ncol(trait_data) - 1),
  mapping = aes(color = TRT),
  upper = list(continuous = custom_cor_sig),
  lower = list(continuous = wrap("points", size = 1.2, alpha = 0.6)),
  diag = list(continuous = wrap("densityDiag")),
  columnLabels = traits
) +
  theme(
    strip.text = element_text(size = 14, face = "bold"),
    panel.spacing = unit(0.2, "lines"),
    plot.margin = margin(10, 10, 10, 10),
    panel.border = element_blank()
  )

# ── Add Single Outer Border ──
cor_plot <- cor_plot +
  theme(plot.background = element_rect(color = "black", linewidth = 1.5, fill = NA))

# ── Save Plot ──
ggsave("cor_plot_full_outer_border.tiff",
       plot = cor_plot, width = 16, height = 12, dpi = 400, units = "in")
