# 3. Identify 60 vs 90 DAS Traits
# ==========================================================

traits_90 <- gsub("60$", "90", traits_60)

# ==========================================================
# 4. Plot Function
# ==========================================================
plot_trait <- function(tr, t60, t90, tag_letter) {
  
  # Reshape
  d <- df %>%
    select(all_of(c(t60, t90))) %>%
    pivot_longer(cols = everything(),
                 names_to = "Time",
                 values_to = "Value") %>%
    mutate(Time = ifelse(Time == t60, "60 DAS", "90 DAS"))
  
  # Summary
  s <- d %>%
    group_by(Time) %>%
    summarise(
      mean = mean(Value, na.rm = TRUE),
      sd   = sd(Value, na.rm = TRUE),
      n    = n(),
      se   = sd / sqrt(n),
      .groups = "drop"
    )
  
  # Y limits for significance
  ymax <- max(s$mean + s$se, na.rm = TRUE)
  y1 <- ymax + 0.05 * ymax
  y2 <- ymax + 0.10 * ymax
  
  # T-test
  p <- t.test(Value ~ Time, data = d)$p.value
  sig <- ifelse(p < 0.001, "***",
                ifelse(p < 0.01, "**",
                       ifelse(p < 0.05, "*", "ns")))
  
  # ----------------------------------------------------------
  # Plot
  # ----------------------------------------------------------
  g <- ggplot(s, aes(Time, mean, fill = Time)) +
    
    geom_col(width = 0.55, color = "black", linewidth = 1) +
    geom_errorbar(aes(ymin = mean - se, ymax = mean + se),
                  width = 0.1, linewidth = 0.8) +
    
    # Bracket + significance
    annotate("segment", x = 1, xend = 2, y = y1, yend = y1, size = 1) +
    annotate("segment", x = 1, xend = 1, y = y1, yend = y1 - 0.03*ymax, size = 1) +
    annotate("segment", x = 2, xend = 2, y = y1, yend = y1 - 0.03*ymax, size = 1) +
    annotate("text", x = 1.5, y = y2, label = sig,
             size = 12, fontface = "bold",
             family = "Times New Roman") +
    
    scale_fill_manual(values = c("60 DAS" = "#1F77B4", "90 DAS" = "#FF7F0E")) +
    scale_y_continuous(expand = expansion(mult = c(0, 0.12))) +
    
    labs(title = tr, tag = tag_letter, fill = "Time (DAS)") +
    
    theme_minimal(base_family = "Times New Roman") +
    theme(
      
      # Legend
      legend.position = "bottom",
      legend.title = element_text(size = 26, face = "bold"),
      legend.text  = element_text(size = 28),
      
      # Axis
      axis.title = element_blank(),
      axis.text.x = element_text(size = 0, face = "bold"),
      axis.text.y = element_text(size = 24, face = "bold"),
      
      # Title & tag
      plot.title = element_text(size = 24, face = "bold",
                                hjust = 0, margin = margin(b = -5)),
      plot.tag = element_text(size = 36, face = "bold",
                              hjust = -0.15, vjust = 1),
      
      # Border only left and bottom
      panel.border = element_blank(),
      axis.line.x = element_line(color = "black", size = 0.9),
      axis.line.y = element_line(color = "black", size = 1.1)
    )
  
  return(g)
}

# ---------------------------
# 1. Load Packages
# ---------------------------
library(dplyr)
library(Hmisc)       # rcorr()
library(reshape2)
library(ggplot2)
library(RColorBrewer)

# ---------------------------
# 2. Numeric data preparation
# ---------------------------
df <- data   # replace with your dataframe
num_df <- df %>% select(where(is.numeric))

# Remove zero-variance columns
zero_var <- sapply(num_df, function(x) var(x, na.rm = TRUE) == 0)
num_df <- num_df[, !zero_var]

# ---------------------------
# 3. Compute correlation + p-values
# ---------------------------
corr_res <- rcorr(as.matrix(num_df), type = "pearson")
corr  <- corr_res$r
pval  <- corr_res$P

# ---------------------------
# 4. Melt into long format
# ---------------------------
corr_df <- melt(corr, varnames = c("Trait1", "Trait2"), value.name = "Correlation")
p_df    <- melt(pval, varnames = c("Trait1", "Trait2"), value.name = "pvalue")
plot_df <- left_join(corr_df, p_df, by = c("Trait1", "Trait2"))

# ---------------------------
# 5. Identify significant correlations
# ---------------------------
plot_df <- plot_df %>%
  mutate(Significant = ifelse(pvalue < 0.05, "Yes", "No"),
         CorrGroup = case_when(
           Significant == "Yes" & Correlation > 0 ~ "Positive",
           Significant == "Yes" & Correlation < 0 ~ "Negative",
           TRUE ~ "NS"
         ))

# ---------------------------
# 6. Compute clustering order (using only significant correlations)
# ---------------------------
sig_corr <- corr
sig_corr[pval >= 0.05] <- 0
sig_corr[is.na(sig_corr)] <- 0

# Distance matrix and hierarchical clustering
dist_mat <- as.dist(1 - sig_corr)
clust <- hclust(dist_mat, method = "average")
order_traits <- clust$labels[clust$order]

# Apply clustering order to factors
plot_df$Trait1 <- factor(plot_df$Trait1, levels = order_traits)
plot_df$Trait2 <- factor(plot_df$Trait2, levels = order_traits)


# ---------------------------
# 7. Plot heatmap with customized legend labels
# ---------------------------
heatmap_plot <- ggplot(plot_df, aes(Trait1, Trait2, fill = CorrGroup)) +
  geom_tile(color = "grey90") +
  scale_fill_manual(
    values = c(
      "Positive" = "#1F77B4", 
      "Negative" = "#FF7F0E",  
      "NS" = "white"
    ),
    labels = c(
      "Positive" = "Sig. (+ve)",
      "Negative" = "Sig. (-ve)",
      "NS" = "NS"
    )
  ) +
  theme_minimal(base_size = 14) +
  theme(
    axis.text.x = element_text(angle = 90, vjust = 0.5, hjust = 1, family = "Times New Roman", face = "bold", color = "black"),
    axis.text.y = element_text(family = "Times New Roman", face = "bold", color = "black"),
    axis.title = element_blank(),
    panel.background = element_rect(fill = "white", color = NA),
    plot.background = element_rect(fill = "white", color = NA),
    panel.grid = element_blank()
  ) +
  guides(fill = guide_legend(title = NULL))  # Remove legend title

# Display heatmap
print(heatmap_plot)

# ---------------------------
# 8. Save as PNG
# ---------------------------
ggsave(filename = "correlation_heatmap_clustered.png",
       plot = heatmap_plot,
       width = 10, height = 8, units = "in", dpi = 300)
