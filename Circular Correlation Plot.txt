Circular Correlation Plot

# Save the plot to a PNG file
png("CIRCULAR_CORRELATION.png", width = 33, height = 20, units = "cm", res = 300)

# Adjust margins to accommodate the legend
par(mar = c(5, 5, 5, 10))  # Extra space on the right for the legend

# Draw the circular correlation plot
chordDiagram(
  correlation_matrix, 
  annotationTrack = "grid",        # Add the grid track for rectangular boxes
  transparency = 0.5,
  col = col_fun,                   # Color gradient based on correlation
  grid.col = circlize::rand_color(ncol(correlation_matrix)), # Random sector colors
  annotationTrackHeight = c(0.05, 0.1)  # Height for labels and boxes
)

# Add trait labels on the edge of rectangular boxes, positioned outside the circular box
circos.track(
  track.index = 1, 
  panel.fun = function(x, y) {
    sector.name <- get.cell.meta.data("sector.index")
    circos.text(
      CELL_META$xcenter, CELL_META$ylim[2] + 0.05,  # Adjusted to place text outside
      labels = sector.name, 
      facing = "clockwise", 
      niceFacing = TRUE, 
      adj = c(0, 0.5), 
      cex = 1,         # Increased font size
      font = 1,        # Bold font
      col = "black",   # Label color
      family = "Times New Roman"  # Times New Roman font
    )
  },
  bg.border = NA   # Remove border from the track
)

# Add a color legend closer to the plot
# Add a color legend to the left-middle part of the plot
legend(
  x = -1.5, y = 0,  # Coordinates for left-middle position
  legend = c("-1", "-0.5", "0", "0.5", "1"),  # Include intermediate values
  fill = col_fun(c(-1, -0.5, 0, 0.5, 1)),  # Match colors with the scale
  title = "Pearson\nCorrelation",  # Title with line break for better spacing
  bty = "n",  # Remove the box around the legend
  cex = 0.8,  # Adjust text size
  border = "white",  # Remove border
  text.col = "black",  # Text color
  inset = c(0, 0)  # No additional inset adjustment
)

# Finish saving the plot
dev.off()
