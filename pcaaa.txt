# ── Load Required Libraries ─────────────────────────────
library(FactoMineR)
library(factoextra)
library(ggplot2)
library(ggrepel)
library(dplyr)
library(extrafont)

# ── Optional: Import fonts if not done before ───────────
# font_import(pattern = "Times", prompt = FALSE)
# loadfonts(device = "win")

# ── Load Your Data ──────────────────────────────────────
data <- SAAAAA  # Replace with your dataset
genotypes <- data$GEN

# ── PCA ─────────────────────────────────────────────────
pca_result <- prcomp(data[, 2:13], scale. = TRUE)
var_coords <- get_pca_var(pca_result)$coord

# Identify traits positively correlated with GY
GY_coords <- var_coords["GY", ]
positive_traits <- rownames(var_coords)[var_coords[, 1] * GY_coords[1] + var_coords[, 2] * GY_coords[2] > 0]

# ── Prepare Scores for Genotype Points ──────────────────
scores <- as.data.frame(pca_result$x)
scores$GEN <- genotypes

# ── Plot ────────────────────────────────────────────────
p <- fviz_pca_biplot(
  pca_result,
  geom.ind = "point",
  col.ind = "#1f77b4",
  pointshape = 16,
  pointsize = 5,  # Increased point size
  fill.ind = "#1f77b4",
  col.var = "brown4",
  repel = TRUE,  # Repel labels
  addEllipses = FALSE,
  label = "var",
  labelsize = 7,  # Increased variable label size
  arrowsize = 1.3,  # Thicker vector arrows
  axes = c(1, 2),
  ggtheme = theme_gray(base_family = "Times New Roman")
) +
  coord_fixed(xlim = c(-5, 5), ylim = c(-5, 5)) +
  theme(
    panel.border = element_rect(color = "white", fill = NA, size = 1.5),  # White square border
    panel.grid.major = element_line(color = "gray85", size = 0.5),  # Subtle grid lines
    panel.grid.minor = element_blank(),
    axis.ticks = element_line(color = "black", size = 1),  # Thicker axis ticks

    legend.position = "none",
    plot.title = element_blank(),
    axis.title = element_text(size = 24, family = "Times New Roman", face = "bold"),
    axis.text = element_text(size = 22, family = "Times New Roman")
  )


# ── Print ───────────────────────────────────────────────
print(p)

# ── Save High Resolution ────────────────────────────────
ggsave("pca_biplot_clean.tiff", plot = p, width = 12, height = 12, dpi = 300)
