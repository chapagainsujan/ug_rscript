# ── Load Libraries ─────────────────────────────────────────────
library(FactoMineR)
library(factoextra)
library(ggplot2)
library(dplyr)
library(RColorBrewer)
library(ggrepel)
library(showtext)
library(patchwork)

# ── Font Setup (Times New Roman) ──────────────────────────────
font_add("Times New Roman", regular = "C:/Windows/Fonts/times.ttf")
showtext_auto()
text_size <- 42  # Base font size

# ── Function for PCA Biplot with selective ellipses ───────────
create_pca_biplot <- function(data, gen_col = "GEN", k = 4) {
  trait_data <- data[, -which(names(data) == gen_col)]
  pca_result <- prcomp(trait_data, scale. = TRUE)
  clusters <- kmeans(pca_result$x[, 1:2], centers = k)$cluster
  data$Group <- factor(clusters)
  
  # PCA individual coordinates
  ind_df <- as.data.frame(pca_result$x[, 1:2])
  colnames(ind_df) <- c("Dim1", "Dim2")
  ind_df$GEN <- data[[gen_col]]
  ind_df$Group <- data$Group
  
  # Base plot from fviz_pca_biplot (with no ellipse)
  base_plot <- fviz_pca_biplot(pca_result,
                               label = "var",
                               addEllipses = FALSE,  # Turn off default ellipses
                               col.var = "midnightblue",
                               col.ind = "black",
                               pointshape = 19,
                               pointsize = 2,
                               repel = TRUE,
                               labelsize = 19) +
    geom_text_repel(data = ind_df, aes(x = Dim1, y = Dim2, label = GEN),
                    size = 14, family = "Times New Roman") +
    geom_point(data = ind_df, aes(x = Dim1, y = Dim2, color = Group), size = 2) +
    scale_color_brewer(palette = "Set2") +
    labs(title = NULL) +
    theme_bw(base_family = "Times New Roman") +
    theme(legend.title = element_text(size = text_size),
          legend.text = element_text(size = text_size - 1),
          axis.title = element_text(size = text_size + 8),
          axis.text = element_text(size = text_size),
          plot.title = element_text(size = text_size + 7, hjust = 0.5))
  
  # Add ellipses only for groups with ≥3 points
  ellipse_data <- ind_df %>%
    group_by(Group) %>%
    filter(n() >= 3)
  
  if (nrow(ellipse_data) > 0) {
    base_plot <- base_plot +
      stat_ellipse(data = ellipse_data, aes(x = Dim1, y = Dim2, color = Group), 
                   level = 0.95, linetype = 1, size = 1)
  }
  
  return(base_plot)
}

# ── Load Your Data ─────────────────────────────────────────────
AA <- aa      # Dataset 1
BB <- SAAAAA  # Dataset 2

# ── Create Individual PCA Biplots ─────────────────────────────
plot_a <- create_pca_biplot(AA)
plot_b <- create_pca_biplot(BB)

# ── Combine with Patchwork and Tags ───────────────────────────
final_plot <- plot_a + plot_b +
  plot_annotation(tag_levels = "a") &
  theme(plot.tag = element_text(size = text_size + 30, face = "bold", family = "Times New Roman"))

# ── Save as TIFF ──────────────────────────────────────────────
tiff("Combined_PCA_Biplots.tiff", width = 6400, height = 3000, res = 300)
print(final_plot)
dev.off()
