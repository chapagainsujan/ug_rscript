# ── 1. Load Libraries ─────────────────────────────────────
library(tidyverse)
library(showtext)
library(patchwork)
library(agricolae)
library(multcompView)

# ── 2. Set Font ───────────────────────────────────────────
font_add("Times New Roman", regular = "C:/Windows/Fonts/times.ttf")
showtext_auto()

# ── 3. Read Data ──────────────────────────────────────────
data <- Leaf  # make sure your data is loaded as 'physio'

# ── 4. Define Traits and Labels ───────────────────────────
traits <- c("LL", "LW", "FLA", "RWC", "LS", "WSD", "RDW", "RSWD")
trait_labels <- c("Leaf Length (cm)", "Leaf Width (cm)", "Flag Leaf Area (cm²)", 
                  "Relative Water Content (%)", "Leaf Succulence (g H₂O/cm²)", 
                  "Water Saturation Deficit (%)", "Relative Dry Weight (g)", 
                  "Relative Saturation Deficit (%)")
plot_labels <- LETTERS[1:length(traits)]

plot_list <- list()

# ── 5. Loop Through Traits ────────────────────────────────
for (i in seq_along(traits)) {
  trait <- traits[i]
  label <- trait_labels[i]
  
  # Summary stats
  summary_dat <- data %>%
    group_by(TRT) %>%
    summarise(mean = mean(.data[[trait]], na.rm = TRUE),
              se = sd(.data[[trait]], na.rm = TRUE) / sqrt(n()), .groups = "drop")
  
  # Tukey HSD
  aov_model <- aov(reformulate("TRT", trait), data = data)
  tukey <- HSD.test(aov_model, "TRT", group = TRUE)
  letters_df <- data.frame(TRT = rownames(tukey$groups),
                           Letter = tukey$groups$groups)
  
  # Merge for plotting
  summary_dat <- summary_dat %>%
    left_join(letters_df, by = "TRT")
  
  # Plot
  p <- ggplot(summary_dat, aes(x = TRT, y = mean, fill = TRT)) +
    geom_bar(stat = "identity", position = position_dodge(0.8), width = 0.7) +
    geom_errorbar(aes(ymin = mean - se, ymax = mean + se),
                  width = 0.2, position = position_dodge(0.8)) +
    geom_text(aes(label = Letter), 
              position = position_dodge(0.8), vjust = -1,
              size = 18, family = "Times New Roman") +
    labs(x = NULL, y = label, tag = paste0(plot_labels[i], ".")) +
    scale_fill_manual(values = c("#1f557f", "#f57c28")) +
    scale_y_continuous(expand = expansion(mult = c(0, 0.1))) +
    theme_minimal(base_family = "Times New Roman") +
    theme(
      axis.text.x = element_text(size = 44, color = "black"),
      axis.text.y = element_text(size = 44, color = "black"),
      axis.title.y = element_text(size = 44, face = "bold"),
      legend.position = "none",
      plot.tag = element_text(size = 50, face = "bold"),
      panel.border = element_rect(color = "black", fill = NA, linewidth = 0.5),
      panel.grid = element_blank(),
      axis.ticks.y = element_line(color = "black", linewidth = 0.5),
      axis.ticks.length.y = unit(0.15, "cm"),
      axis.ticks.length.x = unit(0.15, "cm")
    )
  
  plot_list[[i]] <- p
}

# ── 6. Arrange and Save Plots ─────────────────────────────
final_plot <- wrap_plots(plot_list, ncol = 4) + 
  plot_layout(guides = "collect")

ggsave("trait_analysis_plot.tiff", final_plot, 
       width = 14, height = 9, dpi = 300, device = "tiff")
